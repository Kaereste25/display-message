<script>


  // =====================================

  // HIER DEINE SUPABASE-DATEN EINTRAGEN

  // =====================================


  const SUPABASE_URL =

    "https://puyhamcvmweubywijahm.supabase.co";


  const SUPABASE_KEY =

    "sb_publishable_KNPxsxxrHKR3dwggwuxP8w_Wm_QUkxU";



  const client =

    supabase.createClient(

      SUPABASE_URL,

      SUPABASE_KEY

    );



  // =====================================

  // STATUS

  // =====================================


  function showStatus(text) {

    document.getElementById("status").innerText = text;

  }



  // =====================================

  // LOGIN

  // =====================================


  async function login() {


    const email =

      document.getElementById("email").value;


    const password =

      document.getElementById("password").value;



    showStatus("Anmeldung...");



    const { data, error } =

      await client.auth.signInWithPassword({

        email: email,

        password: password

      });



    if (error) {


      showStatus(

        "Login fehlgeschlagen: " +

        error.message

      );


      return;

    }



    showStatus("Angemeldet");


    await showMessageArea();

  }



  // =====================================

  // LOGOUT

  // =====================================


  async function logout() {


    await client.auth.signOut();


    document.getElementById(

      "loginArea"

    ).style.display = "block";


    document.getElementById(

      "messageArea"

    ).style.display = "none";


    showStatus("Abgemeldet");

  }



  // =====================================

  // NACHRICHT LADEN

  // =====================================


  async function showMessageArea() {


    document.getElementById(

      "loginArea"

    ).style.display = "none";


    document.getElementById(

      "messageArea"

    ).style.display = "block";



    const { data, error } =

      await client

        .from("display_message")

        .select("message")

        .eq("id", 1)

        .single();



    if (!error && data) {


      document.getElementById(

        "message"

      ).value = data.message;


    }

  }



  // =====================================

  // NACHRICHT SENDEN

  // =====================================


  async function sendMessage() {


    const message =

      document.getElementById(

        "message"

      ).value.trim();



    if (!message) {

      showStatus(

        "Bitte eine Nachricht eingeben."

      );

      return;

    }



    showStatus(

      "Nachricht wird gesendet..."

    );



    const { data, error } =

      await client

        .from("display_message")

        .update({

          message: message,

          updated_at:

            new Date().toISOString()

        })

        .eq("id", 1)

        .select();



    if (error) {


      showStatus(

        "Fehler: " +

        error.message

      );


      return;

    }



    showStatus(

      "Nachricht gesendet ✓"

    );

  }



  // =====================================

  // BEIM START PRÜFEN

  // =====================================


  async function init() {


    const {

      data: { session }

    } =

      await client.auth.getSession();



    if (session) {


      await showMessageArea();


    }


  }



  init();


</script>
